function! ClosePair(char)
	if getline('.')[col('.') - 1] == a:char

-- 补全原脚本缺失的 ClosePair 函数（括号补全依赖）
function _G.ClosePair(char)
  local line = vim.fn.getline(".")
  local col = vim.fn.col(".")
  if col > #line or line:sub(col,col) ~= char then
    return char
  else
    return ""
  end
end
